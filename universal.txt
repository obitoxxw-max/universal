-- CONFIG SYSTEM (FIXED - 3 PASSES LOADING)
-- ============================================================
local Config = {}
local ConfigRegistry = {}
local CONFIG_INDEX_FILE = "crybby_configs_index.json"

local function getConfigIndex()
    if readfile and isfile and isfile(CONFIG_INDEX_FILE) then
        local s, e = pcall(function() return readfile(CONFIG_INDEX_FILE) end)
        if s then
            local ok, data = pcall(function() return HttpService:JSONDecode(e) end)
            if ok then return data end
        end
    end
    return {}
end

local function saveConfigIndex(index)
    if writefile then
        pcall(function() writefile(CONFIG_INDEX_FILE, HttpService:JSONEncode(index)) end)
    end
end

local function saveConfigWithName(name)
    if not name or name == "" then return false end
    if not writefile then return false end
    local s, e = pcall(function() return HttpService:JSONEncode(Config) end)
    if not s then return false end
    pcall(function() writefile("crybby_config_" .. name .. ".json", e) end)
    local index = getConfigIndex()
    if not table.find(index, name) then
        table.insert(index, name)
        saveConfigIndex(index)
    end
    return true
end

local function loadConfigWithName(name)
    if not name or name == "" then return false end
    if not readfile or not isfile then return false end
    local file = "crybby_config_" .. name .. ".json"
    if not isfile(file) then return false end
    local s, e = pcall(function() return readfile(file) end)
    if not s then return false end
    local ok, data = pcall(function() return HttpService:JSONDecode(e) end)
    if not ok then return false end
    Config = data
    
    -- PASS 1: Restore sliders/numbers FIRST (so toggles use correct values)
    for key, value in pairs(Config) do
        if type(value) == "number" and not string.find(key, "_Key") then
            if ConfigRegistry[key] then
                pcall(function() ConfigRegistry[key](value) end)
            end
        end
    end
    
    -- PASS 2: Restore toggles/booleans (they now have correct slider values)
    for key, value in pairs(Config) do
        if type(value) == "boolean" then
            if ConfigRegistry[key] then
                pcall(function() ConfigRegistry[key](value) end)
            end
        end
    end
    
    -- PASS 3: Restore keybinds visually
    for key, value in pairs(Config) do
        if string.sub(key, -4) == "_Key" and type(value) == "string" then
            if ConfigRegistry[key] then
                pcall(function() ConfigRegistry[key](value) end)
            end
        end
    end
    
    return true
end

local function deleteConfigWithName(name)
    if not name or name == "" then return false end
    if not delfile or not isfile then return false end
    local file = "crybby_config_" .. name .. ".json"
    if not isfile(file) then return false end
    pcall(function() delfile(file) end)
    local index = getConfigIndex()
    for i, n in ipairs(index) do
        if n == name then table.remove(index, i); break end
    end
    saveConfigIndex(index)
    return true
end

local function applyTheme(themeName)
    if not Themes[themeName] then return end
    CurrentTheme = Themes[themeName]
    for _, btn in ipairs(UI_Elements.Buttons) do if btn.Parent then TweenService:Create(btn, TweenInfo.new(0.3), {BackgroundColor3 = CurrentTheme.Primary}):Play() end end
    for _, kb in ipairs(UI_Elements.Keybinds) do if kb.Parent then TweenService:Create(kb, TweenInfo.new(0.3), {BackgroundColor3 = CurrentTheme.Secondary}):Play() end end
    for _, fill in ipairs(UI_Elements.SliderFills) do if fill.Parent then fill.BackgroundColor3 = CurrentTheme.Primary end end
    for _, knob in ipairs(UI_Elements.SliderKnobs) do if knob.Parent and knob:FindFirstChildOfClass("UIStroke") then knob:FindFirstChildOfClass("UIStroke").Color = CurrentTheme.Primary end end
    for _, border in ipairs(UI_Elements.Borders) do if border.Parent then border.Color = CurrentTheme.Border end end
    for _, tab in ipairs(UI_Elements.TabButtons) do if tab.Parent and tab:FindFirstChildOfClass("UIStroke") then tab:FindFirstChildOfClass("UIStroke").Color = CurrentTheme.Primary end end
    for _, title in ipairs(UI_Elements.SectionTitles) do if title.Parent then title.TextColor3 = CurrentTheme.Primary end end
    for _, sep in ipairs(UI_Elements.SectionSeparators) do if sep.Parent then sep.BackgroundColor3 = CurrentTheme.Border end end
    for _, tData in ipairs(UI_Elements.Toggles) do
        if tData.bg.Parent then tData.bg.BackgroundColor3 = tData.state and CurrentTheme.Primary or CurrentTheme.SurfaceLight end
    end
    if mainFrame then mainFrame.BackgroundColor3 = CurrentTheme.BG end
    if mainBorder then mainBorder.Color = CurrentTheme.Border end
    if hudFrame then hudFrame.BackgroundColor3 = CurrentTheme.BG end
    if hudBorder then hudBorder.Color = CurrentTheme.Border end
    if titleBar then titleBar.BackgroundColor3 = CurrentTheme.Surface end
    if titleFix then titleFix.BackgroundColor3 = CurrentTheme.Surface end
    if tabContainer then tabContainer.BackgroundColor3 = CurrentTheme.Surface end
    if contentContainer then contentContainer.BackgroundColor3 = CurrentTheme.BG end
end

local function createTabButton(tab, index)
    local btn = Instance.new("TextButton", tabContainer)
    btn.Size = UDim2.new(1, -20, 0, 38)
    btn.Position = UDim2.new(0, 10, 0, 10 + (index-1)*42)
    btn.BackgroundColor3 = CurrentTheme.SurfaceLight
    btn.BackgroundTransparency = 0.5
    btn.Text = tab.name
    btn.TextColor3 = CurrentTheme.TextDim
    btn.TextSize = 12
    btn.Font = Enum.Font.GothamBold
    btn.BorderSizePixel = 0
    Instance.new("UICorner", btn).CornerRadius = UDim.new(0, 5)
    local stroke = Instance.new("UIStroke", btn)
    stroke.Color = CurrentTheme.Primary
    stroke.Transparency = 1
    stroke.Thickness = 1
    table.insert(UI_Elements.TabButtons, btn)
    return btn, stroke
end

local function selectTab(tabId)
    if currentTab == tabId then return end
    currentTab = tabId
    for id, btn in pairs(tabButtons) do
        local stroke = btn:FindFirstChildOfClass("UIStroke")
        if id == tabId then
            TweenService:Create(btn, TweenInfo.new(0.25), {BackgroundColor3 = CurrentTheme.Primary, BackgroundTransparency = 0}):Play()
            if stroke then TweenService:Create(stroke, TweenInfo.new(0.25), {Transparency = 0}):Play() end
            btn.TextColor3 = Color3.fromRGB(255, 255, 255)
        else
            TweenService:Create(btn, TweenInfo.new(0.25), {BackgroundColor3 = CurrentTheme.SurfaceLight, BackgroundTransparency = 0.5}):Play()
            if stroke then TweenService:Create(stroke, TweenInfo.new(0.25), {Transparency = 1}):Play() end
            btn.TextColor3 = CurrentTheme.TextDim
        end
    end
    for id, content in pairs(tabContents) do content.Visible = (id == tabId) end
end

for i, tab in ipairs(tabs) do
    local btn, stroke = createTabButton(tab, i)
    tabButtons[tab.id] = btn
    local content = Instance.new("ScrollingFrame", contentContainer)
    content.Name = tab.id .. "Content"
    content.Size = UDim2.new(1, 0, 1, 0)
    content.BackgroundTransparency = 1
    content.ScrollBarThickness = 3

    btn.MouseButton1Click:Connect(function() selectTab(tab.id) end)
end

local function createSection(title, parent)
    local section = Instance.new("Frame", parent)
    section.BackgroundTransparency = 1
    section.AutomaticSize = Enum.AutomaticSize.Y
    section.Size = UDim2.new(1, 0, 0, 0)
    local titleLabel = Instance.new("TextLabel", section)
    titleLabel.Size = UDim2.new(1, 0, 0, 24)
    titleLabel.BackgroundTransparency = 1
    titleLabel.Text = title
    titleLabel.TextColor3 = CurrentTheme.Primary
    titleLabel.TextSize = 13
    titleLabel.Font = Enum.Font.GothamBold
    titleLabel.TextXAlignment = Enum.TextXAlignment.Left
    table.insert(UI_Elements.SectionTitles, titleLabel)
    local sep = Instance.new("Frame", section)
    sep.Size = UDim2.new(1, 0, 0, 1)
    sep.Position = UDim2.new(0, 0, 0, 26)
    sep.BackgroundColor3 = CurrentTheme.Border
    sep.BackgroundTransparency = 0.5
    sep.BorderSizePixel = 0
    table.insert(UI_Elements.SectionSeparators, sep)
    local container = Instance.new("Frame", section)
    container.Size = UDim2.new(1, 0, 0, 0)
    container.Position = UDim2.new(0, 0, 0, 32)
    container.BackgroundTransparency = 1
    container.AutomaticSize = Enum.AutomaticSize.Y
    local layout = Instance.new("UIListLayout", container)
    layout.Padding = UDim.new(0, 8)
    layout.SortOrder = Enum.SortOrder.LayoutOrder
    return container
end

-- TOGGLE WITH KEYBIND (SMALLER + DOUBLE-CLICK TO RESET)
-- ============================================================
local function createToggle(text, parent, default, callback, configKey)
    local container = Instance.new("Frame", parent)
    container.Size = UDim2.new(1, 0, 0, 40)
    container.BackgroundColor3 = CurrentTheme.Surface
    container.BackgroundTransparency = 0.15
    container.BorderSizePixel = 0
    Instance.new("UICorner", container).CornerRadius = UDim.new(0, 5)
    local label = Instance.new("TextLabel", container)
    label.Size = UDim2.new(1, -135, 1, 0)
    label.Position = UDim2.new(0, 14, 0, 0)
    label.BackgroundTransparency = 1
    label.Text = text
    label.TextColor3 = CurrentTheme.Text
    label.TextSize = 13
    label.Font = Enum.Font.GothamBold
    label.TextXAlignment = Enum.TextXAlignment.Left
    local switchBg = Instance.new("Frame", container)
    switchBg.Size = UDim2.new(0, 50, 0, 28)
    switchBg.Position = UDim2.new(1, -130, 0.5, -14)
    switchBg.BackgroundColor3 = default and CurrentTheme.Primary or CurrentTheme.SurfaceLight
    switchBg.BorderSizePixel = 0
    Instance.new("UICorner", switchBg).CornerRadius = UDim.new(1, 0)
    local switchKnob = Instance.new("Frame", switchBg)
    switchKnob.Size = UDim2.new(0, 24, 0, 24)
    switchKnob.Position = default and UDim2.new(1, -26, 0.5, -12) or UDim2.new(0, 2, 0.5, -12)
    switchKnob.BackgroundColor3 = Color3.fromRGB(255, 255, 255)
    switchKnob.BorderSizePixel = 0
    Instance.new("UICorner", switchKnob).CornerRadius = UDim.new(1, 0)
    local knobShadow = Instance.new("UIStroke", switchKnob)
    knobShadow.Color = Color3.fromRGB(0, 0, 0)
    knobShadow.Transparency = 0.85
    knobShadow.Thickness = 1
    local state = default
    local currentKey = nil
    local function setState(val)
        state = val
        local targetPos = state and UDim2.new(1, -26, 0.5, -12) or UDim2.new(0, 2, 0.5, -12)
        local targetColor = state and CurrentTheme.Primary or CurrentTheme.SurfaceLight
        TweenService:Create(switchKnob, TweenInfo.new(0.25, Enum.EasingStyle.Quart), {Position = targetPos}):Play()
        TweenService:Create(switchBg, TweenInfo.new(0.25), {BackgroundColor3 = targetColor}):Play()
        if callback then pcall(callback, state) end
        if configKey then Config[configKey] = state; ConfigRegistry[configKey] = setState end
    end
    local switchClick = Instance.new("TextButton", switchBg)
    switchClick.Size = UDim2.new(1, 0, 1, 0)
    switchClick.BackgroundTransparency = 1
    switchClick.Text = ""
    switchClick.MouseButton1Click:Connect(function() setState(not state) end)
    table.insert(UI_Elements.Toggles, {bg = switchBg, state = state, setState = setState})
    
    -- SMALLER KEYBIND BUTTON (50x22)
    local keyBtn = Instance.new("TextButton", container)
    keyBtn.Size = UDim2.new(0, 50, 0, 22)
    keyBtn.Position = UDim2.new(1, -55, 0.5, -11)
    keyBtn.BackgroundColor3 = CurrentTheme.Secondary
    keyBtn.Text = "None"
    keyBtn.TextColor3 = Color3.fromRGB(255, 255, 255)
    keyBtn.TextSize = 9
    keyBtn.Font = Enum.Font.GothamBold
    keyBtn.BorderSizePixel = 0
    Instance.new("UICorner", keyBtn).CornerRadius = UDim.new(0, 4)
    table.insert(UI_Elements.Keybinds, keyBtn)
    local listening = false
    
    -- DOUBLE-CLICK DETECTION
    local lastClickTime = 0
    
    local function setKeyVisual(keyName)
        keyBtn.Text = keyName or "None"
        currentKey = nil
        -- Try to convert back to KeyCode/UserInputType for functionality
        if keyName and keyName ~= "None" then
            local keyCode = Enum.KeyCode[keyName]
            if keyCode then
                currentKey = keyCode
            else
                local userInput = Enum.UserInputType[keyName]
                if userInput then currentKey = userInput end
            end
        end
    end
    
    keyBtn.MouseButton1Click:Connect(function()
        local now = tick()
        
        -- DOUBLE-CLICK DETECTION (within 400ms)
        if now - lastClickTime < 0.4 then
            -- Double click - RESET keybind
            currentKey = nil
            keyBtn.Text = "None"
            if configKey then Config[configKey .. "_Key"] = "None" end
            lastClickTime = 0
            return
        end
        lastClickTime = now
        
        if listening then return end
        listening = true
        keyBtn.Text = "..."
        local conn
        conn = UserInputService.InputBegan:Connect(function(input)
            if input.UserInputType == Enum.UserInputType.Keyboard then
                if input.KeyCode == Enum.KeyCode.Backspace or input.KeyCode == Enum.KeyCode.Delete then
                    currentKey = nil
                    keyBtn.Text = "None"
                else
                    currentKey = input.KeyCode
                    keyBtn.Text = currentKey.Name
                end
                listening = false
                if configKey then Config[configKey .. "_Key"] = currentKey and currentKey.Name or "None" end
                conn:Disconnect()
            elseif input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.MouseButton2 or input.UserInputType == Enum.UserInputType.MouseButton3 then
                currentKey = input.UserInputType
                keyBtn.Text = input.UserInputType.Name
                listening = false
                if configKey then Config[configKey .. "_Key"] = currentKey.Name end
                conn:Disconnect()
            end
        end)
    end)
    
    if configKey then 
        Config[configKey] = default
        ConfigRegistry[configKey] = setState
        ConfigRegistry[configKey .. "_Key"] = setKeyVisual
    end
    setState(default)
    
    local obj = { 
        GetState = function() return state end, 
        SetState = setState, 
        GetKey = function() return currentKey end,
        SetKeyVisual = setKeyVisual
    }
    table.insert(ToggleObjects, {text = text, obj = obj, configKey = configKey})
    return obj
end

local function createSlider(text, parent, min, max, default, callback, configKey)
    local container = Instance.new("Frame", parent)
    container.Size = UDim2.new(1, 0, 0, 50)
    container.BackgroundColor3 = CurrentTheme.Surface
    container.BackgroundTransparency = 0.15
    container.BorderSizePixel = 0
    Instance.new("UICorner", container).CornerRadius = UDim.new(0, 5)
    local label = Instance.new("TextLabel", container)
    label.Size = UDim2.new(1, -65, 0, 18)
    label.Position = UDim2.new(0, 14, 0, 8)
    label.BackgroundTransparency = 1
    label.Text = text
    label.TextColor3 = CurrentTheme.Text
    label.TextSize = 12
    label.Font = Enum.Font.GothamBold
    label.TextXAlignment = Enum.TextXAlignment.Left
    local valueLabel = Instance.new("TextLabel", container)
    valueLabel.Size = UDim2.new(0, 55, 0, 18)
    valueLabel.Position = UDim2.new(1, -65, 0, 8)
    valueLabel.BackgroundTransparency = 1
    valueLabel.Text = tostring(default)
    valueLabel.TextColor3 = CurrentTheme.Primary
    valueLabel.TextSize = 12
    valueLabel.Font = Enum.Font.GothamBold
    valueLabel.TextXAlignment = Enum.TextXAlignment.Right
    local sliderBg = Instance.new("Frame", container)
    sliderBg.Size = UDim2.new(1, -28, 0, 6)
    sliderBg.Position = UDim2.new(0, 14, 0, 32)
    sliderBg.BackgroundColor3 = CurrentTheme.SurfaceLight
    sliderBg.BorderSizePixel = 0
    Instance.new("UICorner", sliderBg).CornerRadius = UDim.new(0, 3)
    local sliderFill = Instance.new("Frame", sliderBg)
    sliderFill.Size = UDim2.new((default-min)/(max-min), 0, 1, 0)
    sliderFill.BackgroundColor3 = CurrentTheme.Primary
    sliderFill.BorderSizePixel = 0
    Instance.new("UICorner", sliderFill).CornerRadius = UDim.new(0, 3)
    table.insert(UI_Elements.SliderFills, sliderFill)
    local knob = Instance.new("Frame", sliderBg)
    knob.Size = UDim2.new(0, 14, 0, 14)
    knob.Position = UDim2.new((default-min)/(max-min), -7, 0.5, -7)
    knob.BackgroundColor3 = Color3.fromRGB(255, 255, 255)
    knob.BorderSizePixel = 0
    Instance.new("UICorner", knob).CornerRadius = UDim.new(0.5, 0)
    local knobStroke = Instance.new("UIStroke", knob)
    knobStroke.Color = CurrentTheme.Primary
    knobStroke.Thickness = 1.5
    table.insert(UI_Elements.SliderKnobs, knob)
    local dragging = false
    local function setValue(val)
        val = math.clamp(val, min, max)
        local rel = (val - min) / (max - min)
        sliderFill.Size = UDim2.new(rel, 0, 1, 0)
        knob.Position = UDim2.new(rel, -7, 0.5, -7)
        valueLabel.Text = string.format("%.1f", val)
        if callback then pcall(callback, val) end
        if configKey then Config[configKey] = val; ConfigRegistry[configKey] = setValue end
    end
    local function update(input)
        local rel = math.clamp((input.Position.X - sliderBg.AbsolutePosition.X) / sliderBg.AbsoluteSize.X, 0, 1)
        setValue(min + (max - min) * rel)
    end
    sliderBg.InputBegan:Connect(function(i) if i.UserInputType == Enum.UserInputType.MouseButton1 then dragging = true; update(i) end end)
    UserInputService.InputChanged:Connect(function(i) if dragging and i.UserInputType == Enum.UserInputType.MouseMovement then update(i) end end)
    UserInputService.InputEnded:Connect(function(i) if i.UserInputType == Enum.UserInputType.MouseButton1 then dragging = false end end)
    if configKey then Config[configKey] = default; ConfigRegistry[configKey] = setValue end
    setValue(default)
end

local flySec = createSection("Flight", tabContents.Physics)
local flySpeed = 50
local flyConn, noclipConn
local flyTog = createToggle("Fly & Noclip", flySec, false, function(s)
    local char = player.Character
    if not char then return end
    local root = char:FindFirstChild("HumanoidRootPart")
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not root or not hum then return end
    if s then
        hum.PlatformStand = true
        local bv = Instance.new("BodyVelocity", root); bv.MaxForce = Vector3.new(1e5, 1e5, 1e5)
        local bg = Instance.new("BodyGyro", root); bg.MaxTorque = Vector3.new(1e5, 1e5, 1e5); bg.CFrame = root.CFrame
        flyConn = RunService.RenderStepped:Connect(function()
            if not root then return end
            local cam = workspace.CurrentCamera
            local dir = Vector3.new()
            if UserInputService:IsKeyDown(Enum.KeyCode.W) then dir = dir + cam.CFrame.LookVector end
            if UserInputService:IsKeyDown(Enum.KeyCode.S) then dir = dir - cam.CFrame.LookVector end
            if UserInputService:IsKeyDown(Enum.KeyCode.A) then dir = dir - cam.CFrame.RightVector end
            if UserInputService:IsKeyDown(Enum.KeyCode.D) then dir = dir + cam.CFrame.RightVector end
            if UserInputService:IsKeyDown(Enum.KeyCode.Space) then dir = dir + Vector3.new(0, 1, 0) end
            if UserInputService:IsKeyDown(Enum.KeyCode.LeftShift) then dir = dir - Vector3.new(0, 1, 0) end
            bv.Velocity = dir * flySpeed
            bg.CFrame = cam.CFrame
        end)
        noclipConn = RunService.Stepped:Connect(function()
            if not player.Character then return end
            for _, part in ipairs(player.Character:GetDescendants()) do if part:IsA("BasePart") then part.CanCollide = false end end
        end)
    else
        if flyConn then flyConn:Disconnect() end
        if noclipConn then noclipConn:Disconnect() end
        if root:FindFirstChild("BodyVelocity") then root.BodyVelocity:Destroy() end
        if root:FindFirstChild("BodyGyro") then root.BodyGyro:Destroy() end
        hum.PlatformStand = false
        if player.Character then for _, part in ipairs(player.Character:GetDescendants()) do if part:IsA("BasePart") then part.CanCollide = true end end end
    end
end, "Fly")

local espSec = createSection("ESP System", tabContents.ESP)
local espActive = false
local currentEspColor = Color3.fromRGB(255, 85, 105)
local espHighlights, espBillboards, espConnections = {}, {}, {}

local function applyESPToPlayer(p)
    if p == player or not espActive or not p.Character then return end
    if espHighlights[p] then pcall(function() espHighlights[p]:Destroy() end); espHighlights[p] = nil end
    if espBillboards[p] then pcall(function() espBillboards[p]:Destroy() end); espBillboards[p] = nil end
    if espConnections[p] then pcall(function() espConnections[p]:Disconnect() end); espConnections[p] = nil end
    local head = p.Character:FindFirstChild("Head") or p.Character:FindFirstChild("HumanoidRootPart")
    local hum = p.Character:FindFirstChildOfClass("Humanoid")
    if head then
        local hl = Instance.new("Highlight")
        hl.Name = "CRYBBY_ESP"; hl.Adornee = p.Character; hl.FillColor = currentEspColor
        hl.FillTransparency = 0.5; hl.OutlineColor = Color3.fromRGB(255, 255, 255); hl.OutlineTransparency = 0
        hl.Parent = p.Character; espHighlights[p] = hl
        local bb = Instance.new("BillboardGui")
        bb.Name = "CRYBBY_ESP_Name"; bb.Size = UDim2.new(0, 250, 0, 50); bb.StudsOffset = Vector3.new(0, 3, 0)
        bb.AlwaysOnTop = true; bb.Adornee = head; bb.Parent = p.Character
        local txt = Instance.new("TextLabel", bb)
        txt.Size = UDim2.new(1, 0, 1, 0); txt.BackgroundTransparency = 1; txt.TextColor3 = currentEspColor
        txt.TextStrokeTransparency = 0; txt.TextStrokeColor3 = Color3.fromRGB(0, 0, 0); txt.TextSize = 15; txt.Font = Enum.Font.GothamBold
        espBillboards[p] = bb
        local function upd()
            if not hum or not hum.Parent then txt.Text = p.Name .. " | ?"; return end
            txt.Text = p.Name .. " | " .. math.clamp(math.floor(hum.Health), 0, math.floor(hum.MaxHealth)) .. "/" .. math.floor(hum.MaxHealth)
        end
        upd()
        if hum then espConnections[p] = hum.HealthChanged:Connect(upd) end
    end
end

local function setupPlayerESP(p)
    if p == player then return end
    if p.Character then task.spawn(function() task.wait(0.3); applyESPToPlayer(p) end) end
    p.CharacterAdded:Connect(function(char) task.spawn(function() task.wait(0.5); if espActive then applyESPToPlayer(p) end end) end)
end

createToggle("Enable ESP", espSec, false, function(state)
    espActive = state
    if state then for _, p in ipairs(Players:GetPlayers()) do setupPlayerESP(p) end
    else
        for _, hl in pairs(espHighlights) do pcall(function() hl:Destroy() end) end
        for _, bb in pairs(espBillboards) do pcall(function() bb:Destroy() end) end
        for _, conn in pairs(espConnections) do pcall(function() conn:Disconnect() end) end
        espHighlights, espBillboards, espConnections = {}, {}, {}
    end
end, "ESP")

Players.PlayerAdded:Connect(function(p) if espActive then setupPlayerESP(p) end end)
Players.PlayerRemoving:Connect(function(p)
    if espHighlights[p] then pcall(function() espHighlights[p]:Destroy() end); espHighlights[p] = nil end
end)

local function getClosestTarget()
    local cam = workspace.CurrentCamera
    local mousePos = UserInputService:GetMouseLocation()
    local closest, shortest = nil, fovActive and fovRadius or 1200
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= player and p.Character and p.Character.Parent then
            if teamCheck and p.Team and player.Team and p.Team == player.Team then continue end
            local part = p.Character:FindFirstChild("Head") or p.Character:FindFirstChild("HumanoidRootPart")
            local hum = p.Character:FindFirstChildOfClass("Humanoid")
            if part and hum and hum.Health > 0 then
                if wallCheck then
                    local origin = cam.CFrame.Position
                    local dir = (part.Position - origin).Unit
                    local dist = (part.Position - origin).Magnitude
                    local params = RaycastParams.new()
                    params.FilterDescendantsInstances = {player.Character, p.Character}
                    params.FilterType = Enum.RaycastFilterType.Exclude
                    local res = workspace:Raycast(origin, dir * dist, params)
                    if res and not res.Instance:IsDescendantOf(p.Character) then continue end
                end
                local sp, on = cam:WorldToViewportPoint(part.Position)
                if on then
                    local dist = (Vector2.new(sp.X, sp.Y) - mousePos).Magnitude
                    if dist <= shortest then shortest = dist; closest = p end
                end
            end
        end
    end
    return closest
end

UserInputService.InputBegan:Connect(function(input, gp)
    if gp then return end
    if aimbotHoldMode and aimbotTog.GetKey() and (input.KeyCode == aimbotTog.GetKey() or input.UserInputType == aimbotTog.GetKey()) then
        isHoldingAim = true
    end
end)
UserInputService.InputEnded:Connect(function(input, gp)
    if aimbotHoldMode and aimbotTog.GetKey() and (input.KeyCode == aimbotTog.GetKey() or input.UserInputType == aimbotTog.GetKey()) then
        isHoldingAim = false; lockedTarget = nil
    end
end)

RunService.RenderStepped:Connect(function()
    fovCircle.Position = UserInputService:GetMouseLocation()
    fovCircle.Radius = fovRadius
    if aimbotActive then
        if aimbotHoldMode and not isHoldingAim then lockedTarget = nil; return end
        local closest = getClosestTarget()
        if closest then lockedTarget = closest end
        if lockedTarget and lockedTarget.Character and lockedTarget.Character.Parent then
            local targetPart = lockedTarget.Character:FindFirstChild("Head") or lockedTarget.Character:FindFirstChild("HumanoidRootPart")
            if targetPart then
                local targetPos = targetPart.Position
                if prediction > 0 then targetPos = targetPos + ((targetPart.Velocity or Vector3.new()) * (prediction / 1000)) end
                local cam = workspace.CurrentCamera
                if smoothness > 0 then
                    cam.CFrame = cam.CFrame:Lerp(CFrame.new(cam.CFrame.Position, targetPos), (100 - smoothness) / 100)
                else
                    cam.CFrame = CFrame.new(cam.CFrame.Position, targetPos)
                end
            else lockedTarget = nil end
        end
    end
end)

local spyActive, spyConnections = false, {}
local function startSpy()
    if spyActive then return end
    spyActive = true
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= player then table.insert(spyConnections, p.Chatted:Connect(function(msg) addChatMessage(p.Name, msg) end)) end
    end
    table.insert(spyConnections, Players.PlayerAdded:Connect(function(p) table.insert(spyConnections, p.Chatted:Connect(function(msg) addChatMessage(p.Name, msg) end)) end))
    table.insert(spyConnections, player.Chatted:Connect(function(msg) addChatMessage(player.Name .. " (You)", msg) end))
    addChatMessage("SYSTEM", "Chat Spy activated - Listening...")
end
local function stopSpy()
    if not spyActive then return end
    spyActive = false
    for _, conn in ipairs(spyConnections) do pcall(function() conn:Disconnect() end) end
    spyConnections = {}
    addChatMessage("SYSTEM", "Chat Spy deactivated")
end

local function searchCharacter(username)
    if not username or username == "" then return end
    charInfoLabel.Text = "Searching..."
    local userId = nil
    local success, err = pcall(function() userId = Players:GetUserIdFromNameAsync(username) end)
    if not success or not userId then
        charNameLabel.Text = "User not found"
        charInfoLabel.Text = tostring(err) or "Invalid username"
        charThumbnail.Image = ""
        lastSearchedUserId = nil
        return
    end
    lastSearchedUserId = userId
    local thumbSuccess, thumbnail = pcall(function() return Players:GetUserThumbnailAsync(userId, Enum.ThumbnailType.AvatarBust, Enum.ThumbnailSize.Size420x420) end)
    if thumbSuccess and thumbnail then charThumbnail.Image = thumbnail else charThumbnail.Image = "rbxasset://textures/ui/GuiImagePlaceholder.png" end
    charNameLabel.Text = username
    local targetPlayer = Players:FindFirstChild(username)
    if targetPlayer then
        local ping = math.floor((targetPlayer:GetNetworkPing() or 0) * 1000)
        local hp, maxHp = 100, 100
        if targetPlayer.Character then
            local h = targetPlayer.Character:FindFirstChildOfClass("Humanoid")
            if h then hp, maxHp = math.floor(h.Health), math.floor(h.MaxHealth) end
        end
        charInfoLabel.Text = "In-Server | " .. ping .. "ms | " .. hp .. "/" .. maxHp
    else
        charInfoLabel.Text = "Offline | ID: " .. userId .. " | Click 'Apply Skin' to copy"
    end
end

-- ============================================================
-- WINDOW MANAGEMENT
-- ============================================================
local isMinimized = false
minBtn.MouseButton1Click:Connect(function()
    isMinimized = not isMinimized
    tabContainer.Visible, contentContainer.Visible = not isMinimized, not isMinimized
    minBtn.Text = isMinimized and "+" or "-"
    TweenService:Create(mainFrame, TweenInfo.new(0.3), {Size = isMinimized and UDim2.new(0, 840, 0, 52) or UDim2.new(0, 840, 0, 620)}):Play()
end)

local function toggleGui(show)
    show = show == nil and not mainFrame.Visible or show
    mainFrame.Visible = show
end

local dragging, dragStart, startPos = false, nil, nil
titleBar.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 then
        dragging, dragStart, startPos = true, input.Position, mainFrame.Position
    end
end)
UserInputService.InputChanged:Connect(function(input)
    if dragging and input.UserInputType == Enum.UserInputType.MouseMovement then
        local delta = input.Position - dragStart
        mainFrame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
    end
end)
UserInputService.InputEnded:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 then dragging = false end
end)

UserInputService.InputBegan:Connect(function(input, gp)
    if gp then return end
    if guiToggleKey and (input.KeyCode == guiToggleKey or input.UserInputType == guiToggleKey) then toggleGui() end
    for _, tData in ipairs(ToggleObjects) do
        local key = tData.obj.GetKey()
        if key and (input.KeyCode == key or input.UserInputType == key) then
            tData.obj.SetState(not tData.obj.GetState())
        end
    end
end)

closeBtn.MouseButton1Click:Connect(function() toggleGui(false) end)
