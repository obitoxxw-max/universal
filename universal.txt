local function ApplyAnimations(Animations, Character)
    local Animate = Character:FindFirstChild('Animate') or Character:WaitForChild('Animate', 5)
    if not Animate then return end
    for Animation, Ids in next, Animations do
        local AnimObject = Animate:FindFirstChild(Animation)
        if not AnimObject then continue end
        for i, id in next, Ids do
            local AnimationObject
            if Animation == 'idle' then
                AnimationObject = AnimObject:FindFirstChild('Animation'..tostring(i))
            elseif Animation == 'swim' and i == 2 then 
                local swimidle = Animate:FindFirstChild('swimidle')
                AnimationObject = swimidle and swimidle:FindFirstChild('SwimIdle')
            else
                local fullstr = string.upper(string.sub(Animation,1,1))..string.sub(Animation,2,#Animation)
                AnimationObject = AnimObject:FindFirstChild(fullstr..'Anim') or AnimObject:FindFirstChild(fullstr)
            end
            if AnimationObject and id then
                AnimationObject.AnimationId = "rbxassetid://"..tostring(id)
            end
        end
    end
end

-- ROBUST FLING SYSTEM
-- ============================================================
local function SkidFling(TargetPlayer)
    local Character = player.Character
    local Humanoid = Character and Character:FindFirstChildOfClass("Humanoid")
    local RootPart = Humanoid and Humanoid.RootPart
    local TCharacter = TargetPlayer.Character
    if not TCharacter or not Character or not RootPart then return end
    
    local THumanoid = TCharacter:FindFirstChildOfClass("Humanoid")
    local TRootPart = THumanoid and THumanoid.RootPart
    if not THumanoid or not TRootPart or THumanoid.Sit then return end
    
    local OldPos = RootPart.CFrame
    local FPDH = Workspace.FallenPartsDestroyHeight
    Workspace.FallenPartsDestroyHeight = 0/0
    
    local BV = Instance.new("BodyVelocity")
    BV.Parent = RootPart
    BV.Velocity = Vector3.new(0, 0, 0)
    BV.MaxForce = Vector3.new(9e9, 9e9, 9e9)
    
    Humanoid:SetStateEnabled(Enum.HumanoidStateType.Seated, false)
    Character.PrimaryPart = RootPart
    
    local flingActive = true
    task.delay(2, function() flingActive = false end)
    
    local Angle = 0
    while flingActive and TCharacter and TCharacter.Parent and TRootPart do
        Angle = Angle + 100
        local pos1 = CFrame.new(TRootPart.Position) * CFrame.new(0, 1.5, 0) * CFrame.Angles(math.rad(Angle), 0, 0)
        RootPart.CFrame = pos1
        Character:SetPrimaryPartCFrame(pos1)
        RootPart.Velocity = Vector3.new(9e7, 9e8, 9e7)
        RootPart.RotVelocity = Vector3.new(9e8, 9e8, 9e8)
        task.wait()
        
        local pos2 = CFrame.new(TRootPart.Position) * CFrame.new(0, -1.5, 0) * CFrame.Angles(math.rad(Angle), 0, 0)
        RootPart.CFrame = pos2
        Character:SetPrimaryPartCFrame(pos2)
        RootPart.Velocity = Vector3.new(9e7, 9e8, 9e7)
        RootPart.RotVelocity = Vector3.new(9e8, 9e8, 9e8)
        task.wait()
    end
    
    BV:Destroy()
    Humanoid:SetStateEnabled(Enum.HumanoidStateType.Seated, true)
    Workspace.FallenPartsDestroyHeight = FPDH
    
    RootPart.CFrame = OldPos * CFrame.new(0, 0.5, 0)
    Character:SetPrimaryPartCFrame(OldPos * CFrame.new(0, 0.5, 0))
    Humanoid:ChangeState(Enum.HumanoidStateType.GettingUp)
    for _, part in pairs(Character:GetChildren()) do
        if part:IsA("BasePart") then part.Velocity, part.RotVelocity = Vector3.new(), Vector3.new() end
    end
end

local function getTracerOrigin()
    local size = camera.ViewportSize
    local mouse = UserInputService:GetMouseLocation()
    if tracerOrigin == "Bottom" then return Vector2.new(size.X/2, size.Y)
    elseif tracerOrigin == "Top" then return Vector2.new(size.X/2, 0)
    elseif tracerOrigin == "Center" then return size/2
    elseif tracerOrigin == "Mouse" then return mouse
    end
end

local function updateCrosshair(mousePos)
    if not crosshairEnabled then
        for _, l in pairs(crosshairDrawings) do l.Visible = false end
        UserInputService.MouseIconEnabled = true
        return
    end
    
    UserInputService.MouseIconEnabled = false
    
    local gap, size, thick, color = crosshairGap, 10, crosshairThickness, crosshairColor
    
    if crosshairStyle == "Cross" then
        crosshairDrawings.dot.Visible = false
        crosshairDrawings.top.Visible, crosshairDrawings.top.From, crosshairDrawings.top.To, crosshairDrawings.top.Color, crosshairDrawings.top.Thickness = true, Vector2.new(mousePos.X, mousePos.Y - gap), Vector2.new(mousePos.X, mousePos.Y - gap - size), color, thick
        crosshairDrawings.bottom.Visible, crosshairDrawings.bottom.From, crosshairDrawings.bottom.To, crosshairDrawings.bottom.Color, crosshairDrawings.bottom.Thickness = true, Vector2.new(mousePos.X, mousePos.Y + gap), Vector2.new(mousePos.X, mousePos.Y + gap + size), color, thick
        crosshairDrawings.left.Visible, crosshairDrawings.left.From, crosshairDrawings.left.To, crosshairDrawings.left.Color, crosshairDrawings.left.Thickness = true, Vector2.new(mousePos.X - gap, mousePos.Y), Vector2.new(mousePos.X - gap - size, mousePos.Y), color, thick
        crosshairDrawings.right.Visible, crosshairDrawings.right.From, crosshairDrawings.right.To, crosshairDrawings.right.Color, crosshairDrawings.right.Thickness = true, Vector2.new(mousePos.X + gap, mousePos.Y), Vector2.new(mousePos.X + gap + size, mousePos.Y), color, thick
    elseif crosshairStyle == "Circle" then
        for _, l in pairs(crosshairDrawings) do if typeof(l) == "table" or l.Name ~= "dot" then l.Visible = false end end
        crosshairDrawings.dot.Visible, crosshairDrawings.dot.Position, crosshairDrawings.dot.Radius, crosshairDrawings.dot.Color, crosshairDrawings.dot.Thickness, crosshairDrawings.dot.Filled = true, mousePos, gap + 5, color, thick, false
    elseif crosshairStyle == "Dot" then
        for _, l in pairs(crosshairDrawings) do if typeof(l) == "table" or l.Name ~= "dot" then l.Visible = false end end
        crosshairDrawings.dot.Visible, crosshairDrawings.dot.Position, crosshairDrawings.dot.Radius, crosshairDrawings.dot.Color, crosshairDrawings.dot.Thickness, crosshairDrawings.dot.Filled = true, mousePos, 2, color, 1, true
    end
end

    trackFill.Size = UDim2.new((default - min) / (max - min), 0, 1, 0)
    trackFill.BackgroundColor3 = Color3.fromRGB(255, 50, 50)
    trackFill.BorderSizePixel = 0
    trackFill.Parent = trackBg
    Instance.new("UICorner", trackFill).CornerRadius = UDim.new(1, 0)
    
    local dragging = false
    local function setValue(val)
        val = math.clamp(val, min, max)
        local ratio = (val - min) / (max - min)
        TweenService:Create(trackFill, TweenInfo.new(0.1, Enum.EasingStyle.Linear), {Size = UDim2.new(ratio, 0, 1, 0)}):Play()
        valueLabel.Text = tostring(math.floor(val))
        if callback then callback(val) end
    end
    
    trackBg.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 then
            dragging = true
            local rel = math.clamp((input.Position.X - trackBg.AbsolutePosition.X) / trackBg.AbsoluteSize.X, 0, 1)
            setValue(min + (max - min) * rel)
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if dragging and input.UserInputType == Enum.UserInputType.MouseMovement then
            local rel = math.clamp((input.Position.X - trackBg.AbsolutePosition.X) / trackBg.AbsoluteSize.X, 0, 1)
            setValue(min + (max - min) * rel)
        end
    end)
    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 then dragging = false end
    end)
    
    setValue(default)
    return {setValue = setValue}
end

local function saveConfig(configName)
    if not configName or configName == "" then return end

            aimEnabled = aimEnabled, fovRadius = fovRadius, smoothnessVal = smoothnessVal,
            checkWalls = checkWalls, checkTeam = checkTeam, espEnabled = espEnabled,
            showNames = showNames, showHealth = showHealth, espStyle = espStyle,
            speedEnabled = speedEnabled, walkSpeedVal = walkSpeedVal, flyEnabled = flyEnabled, 
            flySpeedVal = flySpeedVal, noclipEnabled = noclipEnabled, infJumpEnabled = infJumpEnabled, 
            antiFlingEnabled = antiFlingEnabled, antiAFK = antiAFK, fullbright = fullbright,

        }))
        refreshConfigList()
        renderConfigTable()
        configStatusText = "Saved: " .. configName
        if configStatusLabel then configStatusLabel.Text = "Status: " .. configStatusText end
    end)
end

local function isAimKeyHeld()
    if typeof(aimKey) == "EnumItem" then
        if aimKey.EnumType == Enum.KeyCode then return UserInputService:IsKeyDown(aimKey)
        elseif aimKey.EnumType == Enum.UserInputType then return UserInputService:IsMouseButtonPressed(aimKey) end
    end
    return false
end

local function cleanupESP(p)
    local data = activeESPs[p]
    if data then
        pcall(function()
            if data.Highlight then data.Highlight:Destroy() end
            if data.Billboard then data.Billboard:Destroy() end
            if data.SelectionBox then data.SelectionBox:Destroy() end
            if data.SkeletonLines then
                for _, lineData in pairs(data.SkeletonLines) do
                    if lineData.drawing and lineData.drawing.Remove then 
                        lineData.drawing:Remove() 
                    end
                end
            end
        end)
        activeESPs[p] = nil
    end
end

local function ensureESP(p)
    if p == player or not espEnabled then return end
    local char = p.Character
    if not char then return end
    
    local needsRecreation = false
    if activeESPs[p] then
        if activeESPs[p].Style ~= espStyle then
            needsRecreation = true
        end
    else
        needsRecreation = true
    end

    if needsRecreation then
        cleanupESP(p)
        local data = { Style = espStyle }
        
        local root = char:FindFirstChild("HumanoidRootPart")
        local head = char:FindFirstChild("Head")
        local hum = char:FindFirstChildOfClass("Humanoid")
        if root then
            local bb = Instance.new("BillboardGui")

-- MAIN RENDER LOOP (AIMLOCK CORRIGÉ + ESP + CROSSHAIR)
-- ============================================================
RunService.RenderStepped:Connect(function()
    camera = Workspace.CurrentCamera
    if not camera then return end
    local mousePos = UserInputService:GetMouseLocation()
    
    -- Positionnement et couleur du FOV
    fovCircle.Position = mousePos
    fovCircle.Color = fovColor
    
    updateCrosshair(mousePos)
    
    -- ============================================================
    -- AIMLOCK LOGIC (ENTIÈREMENT RESTAURÉE)
    -- ============================================================
    if aimEnabled and isAimKeyHeld() then
        if lockedTarget and lockedTarget.Character then
            local hum = lockedTarget.Character:FindFirstChildOfClass("Humanoid")
            if hum and hum.Health > 0 then
                local targetPart = lockedTarget.Character:FindFirstChild("Head") or lockedTarget.Character:FindFirstChild("HumanoidRootPart")
                if targetPart then
                    local valid = true
                    if checkTeam and player.Team and lockedTarget.Team and player.Team == lockedTarget.Team then valid = false end
                    if checkWalls and valid then
                        local origin = camera.CFrame.Position
                        local direction = (targetPart.Position - origin).Unit
                        local distance = (targetPart.Position - origin).Magnitude
                        local params = RaycastParams.new()
                        params.FilterDescendantsInstances = {player.Character, lockedTarget.Character}
                        params.FilterType = Enum.RaycastFilterType.Exclude
                        local rayResult = Workspace:Raycast(origin, direction * distance, params)
                        if rayResult and not rayResult.Instance:IsDescendantOf(lockedTarget.Character) then valid = false end
                    end
                    if valid then
                        local targetPos = targetPart.Position
                        if smoothnessVal <= 1 then 
                            camera.CFrame = CFrame.new(camera.CFrame.Position, targetPos)
                        else 
                            camera.CFrame = camera.CFrame:Lerp(CFrame.new(camera.CFrame.Position, targetPos), (21 - smoothnessVal) / 100) 
                        end
                    else
                        lockedTarget = nil
                    end
                end
            else
                lockedTarget = nil
            end
        end
        
        if not lockedTarget then
            local closestTarget, shortestDistance = nil, fovRadius
            for _, p in ipairs(Players:GetPlayers()) do
                if p == player then continue end
                if #targetedPlayers > 0 then
                    local isTargeted = false
                    for _, tName in ipairs(targetedPlayers) do if tName == p.Name then isTargeted = true break end end
                    if not isTargeted then continue end
                end
                if checkTeam and p.Team and player.Team and p.Team == player.Team then continue end
                local char = p.Character
                if not char then continue end
                local hum = char:FindFirstChildOfClass("Humanoid")
                if not hum or hum.Health <= 0 then continue end
                local targetPart = char:FindFirstChild("Head") or char:FindFirstChild("HumanoidRootPart")
                if not targetPart then continue end
                if checkWalls then
                    local origin = camera.CFrame.Position
                    local direction = (targetPart.Position - origin).Unit
                    local distance = (targetPart.Position - origin).Magnitude
                    local params = RaycastParams.new()
                    params.FilterDescendantsInstances = {player.Character, char}
                    params.FilterType = Enum.RaycastFilterType.Exclude
                    local rayResult = Workspace:Raycast(origin, direction * distance, params)
                    if rayResult and not rayResult.Instance:IsDescendantOf(char) then continue end
                end
                local screenPos, onScreen = camera:WorldToViewportPoint(targetPart.Position)
                if onScreen then
                    local dist = (Vector2.new(screenPos.X, screenPos.Y) - mousePos).Magnitude
                    if dist < shortestDistance then 
                        shortestDistance = dist
                        closestTarget = p 
                    end
                end
            end
            if closestTarget then
                lockedTarget = closestTarget
                local tp = lockedTarget.Character:FindFirstChild("Head") or lockedTarget.Character:FindFirstChild("HumanoidRootPart")
                if tp then
                    local targetPos = tp.Position
                    if smoothnessVal <= 1 then 
                        camera.CFrame = CFrame.new(camera.CFrame.Position, targetPos)
                    else 
                        camera.CFrame = camera.CFrame:Lerp(CFrame.new(camera.CFrame.Position, targetPos), (21 - smoothnessVal) / 100) 
                    end
                end
            end
        end
    else
        lockedTarget = nil
    end
    
    -- ============================================================
    -- ESP LOGIC
    -- ============================================================
    if espEnabled then
        for _, p in ipairs(Players:GetPlayers()) do
            ensureESP(p)
            local data = activeESPs[p]
            if not data or not p.Character then continue end
            
            local char = p.Character
            local hum = char:FindFirstChildOfClass("Humanoid")
            if data.Billboard and hum then
                local hl = data.Billboard:FindFirstChild("HL")
                if hl then hl.Text = math.floor(hum.Health).."/"..math.floor(hum.MaxHealth) end
            end
            
            if espStyle == "Skeleton" and data.SkeletonLines then
                if not char:FindFirstChild("HumanoidRootPart") then
                    for _, lineData in pairs(data.SkeletonLines) do 
                        lineData.drawing.Visible = false 
                    end
                    continue
                end
                
                local humanoid = char:FindFirstChildOfClass("Humanoid")
                local joints = {}
                local connections = {}
                
                if humanoid and humanoid.RigType == Enum.HumanoidRigType.R15 then
                    joints = {
                        ["Head"] = char:FindFirstChild("Head"),
                        ["UpperTorso"] = char:FindFirstChild("UpperTorso"),
                        ["LowerTorso"] = char:FindFirstChild("LowerTorso"),
                        ["LeftUpperArm"] = char:FindFirstChild("LeftUpperArm"),
                        ["LeftLowerArm"] = char:FindFirstChild("LeftLowerArm"),
                        ["LeftHand"] = char:FindFirstChild("LeftHand"),
                        ["RightUpperArm"] = char:FindFirstChild("RightUpperArm"),
                        ["RightLowerArm"] = char:FindFirstChild("RightLowerArm"),
                        ["RightHand"] = char:FindFirstChild("RightHand"),
                        ["LeftUpperLeg"] = char:FindFirstChild("LeftUpperLeg"),
                        ["LeftLowerLeg"] = char:FindFirstChild("LeftLowerLeg"),
                        ["RightUpperLeg"] = char:FindFirstChild("RightUpperLeg"),
                        ["RightLowerLeg"] = char:FindFirstChild("RightLowerLeg"),
                    }
                    connections = {
                        { "Head", "UpperTorso" }, { "UpperTorso", "LowerTorso" },
                        { "LowerTorso", "LeftUpperLeg" }, { "LeftUpperLeg", "LeftLowerLeg" },
                        { "LowerTorso", "RightUpperLeg" }, { "RightUpperLeg", "RightLowerLeg" },
                        { "UpperTorso", "LeftUpperArm" }, { "LeftUpperArm", "LeftLowerArm" }, { "LeftLowerArm", "LeftHand" },
                        { "UpperTorso", "RightUpperArm" }, { "RightUpperArm", "RightLowerArm" }, { "RightLowerArm", "RightHand" },
                    }
                elseif humanoid and humanoid.RigType == Enum.HumanoidRigType.R6 then
                    joints = {
                        ["Head"] = char:FindFirstChild("Head"),
                        ["Torso"] = char:FindFirstChild("Torso"),
                        ["LeftLeg"] = char:FindFirstChild("Left Leg"),
                        ["RightLeg"] = char:FindFirstChild("Right Leg"),
                        ["LeftArm"] = char:FindFirstChild("Left Arm"),
                        ["RightArm"] = char:FindFirstChild("Right Arm"),
                    }
                    connections = {
                        { "Head", "Torso" }, { "Torso", "LeftArm" }, { "Torso", "RightArm" },
                        { "Torso", "LeftLeg" }, { "Torso", "RightLeg" },
                    }
                end
                
                for index, connection in ipairs(connections) do
                    local jointA = joints[connection[1]]
                    local jointB = joints[connection[2]]
                    
                    if jointA and jointB then
                        local posA, onScreenA = camera:WorldToViewportPoint(jointA.Position)
                        local posB, onScreenB = camera:WorldToViewportPoint(jointB.Position)
                        
                        local targetFrom = Vector2.new(posA.X, posA.Y)
                        local targetTo = Vector2.new(posB.X, posB.Y)
                        
                        local lineData = data.SkeletonLines[index]
                        if not lineData then
                            lineData = {
                                drawing = Drawing.new("Line")
                            }
                            data.SkeletonLines[index] = lineData
                        end
                        
                        lineData.drawing.From = targetFrom
                        lineData.drawing.To = targetTo
                        lineData.drawing.Color = espColor
                        lineData.drawing.Thickness = 1.5
                        lineData.drawing.Transparency = 1
                        
                        if onScreenA and onScreenB then
                            lineData.drawing.Visible = true
                        else
                            lineData.drawing.Visible = false
                        end
                    elseif data.SkeletonLines[index] then
                        data.SkeletonLines[index].drawing.Visible = false
                    end
                end
            end
        end
    else
        for p, _ in pairs(activeESPs) do cleanupESP(p) end
    end
    
    -- Tracers Logic
    for _, p in ipairs(Players:GetPlayers()) do
        if p == player then continue end
        
        local vis = playerVisuals[p]
        if not vis then
            vis = {tracer = Drawing.new("Line")}
            playerVisuals[p] = vis
        end
        
        local char = p.Character
        local hum = char and char:FindFirstChildOfClass("Humanoid")
        local root = char and char:FindFirstChild("HumanoidRootPart")
        
        local valid = char and hum and root and hum.Health > 0
        local teamValid = not checkTeam or not player.Team or not p.Team or player.Team ~= p.Team
        
        if not valid or not teamValid or not espEnabled or not tracersEnabled then
            if vis.tracer then vis.tracer.Visible = false end
            continue
        end
        
        local screenPos, onScreen = camera:WorldToViewportPoint(root.Position)
        local screenVec = Vector2.new(screenPos.X, screenPos.Y)
        
        if onScreen then
            vis.tracer.Visible = true
            vis.tracer.From = getTracerOrigin()
            vis.tracer.To = screenVec
            vis.tracer.Color = espColor
            vis.tracer.Thickness = 1.5
        else
            vis.tracer.Visible = false
        end
    end
end)

local function applyMovement()
    local char = player.Character
    if not char then return end
    local hum = char:FindFirstChildOfClass("Humanoid")
    if hum and not flyEnabled then hum.WalkSpeed = speedEnabled and walkSpeedVal or 16 end
end

function updateFly()
    local char = player.Character
    if not char then return end
    local root = char:FindFirstChild("HumanoidRootPart")
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not root or not hum then return end
    if flyEnabled then
        hum.PlatformStand = true
        if not flyConn then
            flyConn = RunService.RenderStepped:Connect(function()
                if not flyEnabled or not player.Character then updateFly(); return end
                local currentRoot = player.Character:FindFirstChild("HumanoidRootPart")
                if not currentRoot then return end
                local cam = Workspace.CurrentCamera
                local moveDir = Vector3.new(0, 0, 0)
                if UserInputService:IsKeyDown(Enum.KeyCode.W) then moveDir = moveDir + cam.CFrame.LookVector end
                if UserInputService:IsKeyDown(Enum.KeyCode.S) then moveDir = moveDir - cam.CFrame.LookVector end
                if UserInputService:IsKeyDown(Enum.KeyCode.A) then moveDir = moveDir - cam.CFrame.RightVector end
                if UserInputService:IsKeyDown(Enum.KeyCode.D) then moveDir = moveDir + cam.CFrame.RightVector end
                if UserInputService:IsKeyDown(Enum.KeyCode.Space) then moveDir = moveDir + Vector3.new(0, 1, 0) end
                if UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) or UserInputService:IsKeyDown(Enum.KeyCode.LeftShift) then moveDir = moveDir - Vector3.new(0, 1, 0) end
                if moveDir.Magnitude > 0 then moveDir = moveDir.Unit end
                currentRoot.AssemblyLinearVelocity = moveDir * flySpeedVal
                currentRoot.RotVelocity = Vector3.new(0, 0, 0)
            end)
        end
    else
        if flyConn then flyConn:Disconnect(); flyConn = nil end
        hum.PlatformStand = false
        applyMovement()
    end
end

local antiFlingLastCheck = 0
RunService.Stepped:Connect(function()
    local currentTime = tick()
    if antiFlingEnabled and player.Character and currentTime - antiFlingLastCheck > 0.1 then
        antiFlingLastCheck = currentTime
        local root = player.Character:FindFirstChild("HumanoidRootPart")
        local hum = player.Character:FindFirstChildOfClass("Humanoid")
        if root and hum and hum.Health > 0 and not flyEnabled then
            local vel = root.AssemblyLinearVelocity
            if vel.Magnitude > 200 then
                root.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
                root.Velocity = Vector3.new(0, 0, 0)
            end
            for _, p in ipairs(Players:GetPlayers()) do
                if p ~= player and p.Character then
                    local pRoot = p.Character:FindFirstChild("HumanoidRootPart")
                    if pRoot then pRoot.CanCollide = false end
                end
            end
        end
    end
    if noclipEnabled and player.Character then
        for _, part in ipairs(player.Character:GetDescendants()) do
            if part:IsA("BasePart") then part.CanCollide = false end
        end
    end
end)

    if gameProcessed then return end
    local function isKeyMatch(keybind, input)
        if not keybind then return false end
        if typeof(keybind) == "EnumItem" then
            if keybind.EnumType == Enum.KeyCode and input.KeyCode == keybind then return true end
            if keybind.EnumType == Enum.UserInputType and input.UserInputType == keybind then return true end
        end
        return false
    end
    if isKeyMatch(speedKeybind, input) then speedEnabled = not speedEnabled; speedToggle.setState(speedEnabled); applyMovement() end
    if isKeyMatch(flyKeybind, input) then flyEnabled = not flyEnabled; flyToggle.setState(flyEnabled); updateFly() end
    if isKeyMatch(jumpKeybind, input) then infJumpEnabled = not infJumpEnabled; infJumpToggle.setState(infJumpEnabled) end
    if isKeyMatch(noclipKeybind, input) then noclipEnabled = not noclipEnabled; noclipToggle.setState(noclipEnabled) end
end)
