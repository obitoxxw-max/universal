local function applyTheme(themeName)
    if not Themes[themeName] then return end
    CurrentTheme = Themes[themeName]
    
    for _, btn in ipairs(UI_Elements.Buttons) do
        if btn.Parent then TweenService:Create(btn, TweenInfo.new(0.3), {BackgroundColor3 = CurrentTheme.Primary}):Play() end
    end
    for _, kb in ipairs(UI_Elements.Keybinds) do
        if kb.Parent then TweenService:Create(kb, TweenInfo.new(0.3), {BackgroundColor3 = CurrentTheme.Secondary}):Play() end
    end
    for _, fill in ipairs(UI_Elements.SliderFills) do
        if fill.Parent then fill.BackgroundColor3 = CurrentTheme.Primary end
    end
    for _, knob in ipairs(UI_Elements.SliderKnobs) do
        if knob.Parent and knob:FindFirstChildOfClass("UIStroke") then
            knob:FindFirstChildOfClass("UIStroke").Color = CurrentTheme.Primary
        end
    end
    for _, border in ipairs(UI_Elements.Borders) do
        if border.Parent then border.Color = CurrentTheme.Border end
    end
    for _, tab in ipairs(UI_Elements.TabButtons) do
        if tab.Parent and tab:FindFirstChildOfClass("UIStroke") then
            tab:FindFirstChildOfClass("UIStroke").Color = CurrentTheme.Primary
        end
    end
    for _, title in ipairs(UI_Elements.SectionTitles) do
        if title.Parent then title.TextColor3 = CurrentTheme.Primary end
    end
    for _, sep in ipairs(UI_Elements.SectionSeparators) do
        if sep.Parent then sep.BackgroundColor3 = CurrentTheme.Border end
    end
    
    if mainFrame then mainFrame.BackgroundColor3 = CurrentTheme.BG end
    if mainBorder then mainBorder.Color = CurrentTheme.Border end
    if hudFrame then hudFrame.BackgroundColor3 = CurrentTheme.BG end
    if hudBorder then hudBorder.Color = CurrentTheme.Border end
    if titleBar then titleBar.BackgroundColor3 = CurrentTheme.Surface end
    if titleFix then titleFix.BackgroundColor3 = CurrentTheme.Surface end
    if tabContainer then tabContainer.BackgroundColor3 = CurrentTheme.Surface end
    if contentContainer then contentContainer.BackgroundColor3 = CurrentTheme.BG end
end

local function selectTab(tabId)
    if currentTab == tabId then return end
    currentTab = tabId
    
    for id, btn in pairs(tabButtons) do
        local stroke = btn:FindFirstChildOfClass("UIStroke")
        if id == tabId then
            TweenService:Create(btn, TweenInfo.new(0.25), {
                BackgroundColor3 = CurrentTheme.Primary,
                BackgroundTransparency = 0
            }):Play()
            if stroke then TweenService:Create(stroke, TweenInfo.new(0.25), {Transparency = 0}):Play() end
            btn.TextColor3 = Color3.fromRGB(255, 255, 255)
        else
            TweenService:Create(btn, TweenInfo.new(0.25), {
                BackgroundColor3 = CurrentTheme.SurfaceLight,
                BackgroundTransparency = 0.5
            }):Play()
            if stroke then TweenService:Create(stroke, TweenInfo.new(0.25), {Transparency = 1}):Play() end
            btn.TextColor3 = CurrentTheme.TextDim
        end
    end
    
    for id, content in pairs(tabContents) do
        content.Visible = (id == tabId)
    end
end

for i, tab in ipairs(tabs) do
    local btn, stroke = createTabButton(tab, i)
    tabButtons[tab.id] = btn
    
    local content = Instance.new("ScrollingFrame", contentContainer)

local function createToggle(text, parent, default, callback, onKeyChanged)
    local container = Instance.new("Frame", parent)
    container.Size = UDim2.new(1, 0, 0, 38)
    container.BackgroundColor3 = CurrentTheme.Surface
    container.BackgroundTransparency = 0.15
    container.BorderSizePixel = 0
    Instance.new("UICorner", container).CornerRadius = UDim.new(0, 5)
    
    local label = Instance.new("TextLabel", container)
    label.Size = UDim2.new(1, -85, 1, 0)
    label.Position = UDim2.new(0, 14, 0, 0)
    label.BackgroundTransparency = 1
    label.Text = text
    label.TextColor3 = CurrentTheme.Text
    label.TextSize = 13
    label.Font = Enum.Font.GothamBold
    label.TextXAlignment = Enum.TextXAlignment.Left
    
    local toggleBtn = Instance.new("TextButton", container)
    toggleBtn.Size = UDim2.new(0, 60, 1, -10)
    toggleBtn.Position = UDim2.new(1, -70, 0, 5)
    toggleBtn.BackgroundColor3 = CurrentTheme.SurfaceLight
    toggleBtn.Text = "OFF"
    toggleBtn.TextColor3 = Color3.fromRGB(225, 85, 105)
    toggleBtn.TextSize = 12
    toggleBtn.Font = Enum.Font.GothamBold
    toggleBtn.BorderSizePixel = 0
    Instance.new("UICorner", toggleBtn).CornerRadius = UDim.new(0, 4)
    
    local state = default
    local currentKey = nil
    
    local function setState(val)
        state = val
        toggleBtn.Text = state and "ON" or "OFF"
        toggleBtn.TextColor3 = state and Color3.fromRGB(110, 255, 160) or Color3.fromRGB(225, 85, 105)
        if callback then pcall(callback, state) end
    end
    
    toggleBtn.MouseButton1Click:Connect(function()
        setState(not state)
    end)
    
    if onKeyChanged then
        label.Size = UDim2.new(1, -160, 1, 0)
        toggleBtn.Position = UDim2.new(1, -150, 0, 5)
        
        local keyBtn = Instance.new("TextButton", container)
        keyBtn.Size = UDim2.new(0, 75, 1, -10)
        keyBtn.Position = UDim2.new(1, -70, 0, 5)
        keyBtn.BackgroundColor3 = CurrentTheme.Secondary
        keyBtn.Text = "None"
        keyBtn.TextColor3 = Color3.fromRGB(255, 255, 255)
        keyBtn.TextSize = 11
        keyBtn.Font = Enum.Font.GothamBold
        keyBtn.BorderSizePixel = 0
        Instance.new("UICorner", keyBtn).CornerRadius = UDim.new(0, 4)
        table.insert(UI_Elements.Keybinds, keyBtn)
        
        local listening = false
        
        keyBtn.MouseButton1Click:Connect(function()
            if listening then return end
            listening = true
            keyBtn.Text = "..."
            
            local conn
            conn = UserInputService.InputBegan:Connect(function(input)
                if input.UserInputType == Enum.UserInputType.Keyboard then
                    if input.KeyCode == Enum.KeyCode.Backspace or input.KeyCode == Enum.KeyCode.Delete then
                        currentKey = nil
                        keyBtn.Text = "None"
                    else
                        currentKey = input.KeyCode
                        keyBtn.Text = currentKey.Name
                    end
                    listening = false
                    if onKeyChanged then pcall(onKeyChanged, currentKey) end
                    conn:Disconnect()
                elseif input.UserInputType == Enum.UserInputType.MouseButton1 or 
                       input.UserInputType == Enum.UserInputType.MouseButton2 or 
                       input.UserInputType == Enum.UserInputType.MouseButton3 then
                    currentKey = input.UserInputType
                    keyBtn.Text = input.UserInputType.Name
                    listening = false
                    if onKeyChanged then pcall(onKeyChanged, currentKey) end
                    conn:Disconnect()
                end
            end)
        end)
    end
    
    return {
        GetState = function() return state end,
        SetState = setState,
        GetKey = function() return currentKey end
    }
end

local function isTargetVisible(targetPart, targetChar)
    local cam = workspace.CurrentCamera
    local origin = cam.CFrame.Position
    local targetPos = targetPart.Position
    local direction = (targetPos - origin).Unit
    local distance = (targetPos - origin).Magnitude
    
    local raycastParams = RaycastParams.new()
    raycastParams.FilterDescendantsInstances = {player.Character, targetChar}
    raycastParams.FilterType = Enum.RaycastFilterType.Exclude
    raycastParams.IgnoreWater = true
    
    local result = workspace:Raycast(origin, direction * distance, raycastParams)
    
    if not result then
        return true
    end
    
    if result.Instance and result.Instance:IsDescendantOf(targetChar) then
        return true
    end
    
    return false
end

UserInputService.InputBegan:Connect(function(input, gp)
    if gp then return end
    if aimbotHoldMode and aimbotKey then
        if (input.UserInputType == Enum.UserInputType.Keyboard and input.KeyCode == aimbotKey) or
           (input.UserInputType == aimbotKey) then
            isHoldingAim = true
        end
    end
end)

UserInputService.InputEnded:Connect(function(input, gp)
    if aimbotHoldMode and aimbotKey then
        if (input.UserInputType == Enum.UserInputType.Keyboard and input.KeyCode == aimbotKey) or
           (input.UserInputType == aimbotKey) then
            isHoldingAim = false
            lockedTarget = nil
        end
    end
end)

RunService.RenderStepped:Connect(function()
    local mousePos = UserInputService:GetMouseLocation()
    fovCircle.Position = mousePos
    fovCircle.Radius = fovRadius

    if aimbotActive then
        if aimbotHoldMode and not isHoldingAim then
            lockedTarget = nil
            return
        end

        local cam = workspace.CurrentCamera
        local closestPlayer = nil
        local shortestDist = fovActive and fovRadius or 1200

        for _, p in ipairs(Players:GetPlayers()) do
            if p ~= player and p.Character then
                if teamCheck and p.Team and player.Team and p.Team == player.Team then 
                    continue 
                end
                
                local char = p.Character
                local part = char:FindFirstChild("Head") or 
                            char:FindFirstChild("HumanoidRootPart") or 
                            char:FindFirstChild("Torso") or
                            char:FindFirstChild("UpperTorso")
                
                if part then
                    if wallCheck then
                        if not isTargetVisible(part, char) then
                            continue
                        end
                    end

                    local screenPos, onScreen = cam:WorldToViewportPoint(part.Position)
                    if onScreen then
                        local dist = (Vector2.new(screenPos.X, screenPos.Y) - mousePos).Magnitude
                        if dist <= shortestDist then
                            shortestDist = dist
                            closestPlayer = p
                        end
                    end
                end
            end
        end

        if closestPlayer then
            lockedTarget = closestPlayer
        end

        if lockedTarget and lockedTarget.Character then
            local targetChar = lockedTarget.Character
            local targetPart = targetChar:FindFirstChild("Head") or 
                              targetChar:FindFirstChild("HumanoidRootPart") or 
                              targetChar:FindFirstChild("Torso") or
                              targetChar:FindFirstChild("UpperTorso")
            
            if targetPart then
                local targetPos = targetPart.Position
                
                if prediction > 0 then
                    local vel = targetPart.Velocity or Vector3.new()
                    targetPos = targetPos + (vel * (prediction / 1000))
                end

                if smoothness > 0 then
                    local targetCF = CFrame.new(cam.CFrame.Position, targetPos)
                    cam.CFrame = cam.CFrame:Lerp(targetCF, (100 - smoothness) / 100)
                else
                    cam.CFrame = CFrame.new(cam.CFrame.Position, targetPos)
                end
            else
                lockedTarget = nil
            end
        end
    end
end)

UserInputService.InputBegan:Connect(function(input, gp)
    if gp then return end
    if input.UserInputType == Enum.UserInputType.Keyboard then
        if aimbotKey and input.KeyCode == aimbotKey and not aimbotHoldMode then
            aimbotToggleObj.SetState(not aimbotToggleObj.GetState())
        end
    elseif input.UserInputType == aimbotKey and not aimbotHoldMode then
        aimbotToggleObj.SetState(not aimbotToggleObj.GetState())
    end
end)

            table.insert(spyConnections, conn)
        end
    end
    
    local addedConn = Players.PlayerAdded:Connect(function(p)
        local conn = p.Chatted:Connect(function(message)
            addChatMessage(p.Name, message)
        end)
        table.insert(spyConnections, conn)
    end)
    table.insert(spyConnections, addedConn)
    
    local selfConn = player.Chatted:Connect(function(message)
        addChatMessage(player.Name .. " (You)", message)
    end)
    table.insert(spyConnections, selfConn)
    
    addChatMessage("SYSTEM", "Chat Spy activated - Listening...")
end

local function stopSpy()
    if not spyActive then return end
    spyActive = false
    
    for _, conn in ipairs(spyConnections) do
        pcall(function() conn:Disconnect() end)
    end
    spyConnections = {}
    
    addChatMessage("SYSTEM", "Chat Spy deactivated")
end

local function searchCharacter(username)
    if not username or username == "" then return end
    
    local userId = nil
    pcall(function()
        userId = Players:GetUserIdFromNameAsync(username)
    end)
    
    if not userId then
        charNameLabel.Text = "User not found"
        charInfoLabel.Text = ""
        charThumbnail.Image = ""
        lastSearchedUserId = nil
        return
    end
    
    lastSearchedUserId = userId
    
    -- Récupérer le thumbnail
    local thumbnail = nil
    pcall(function()
        thumbnail = Players:GetUserThumbnailAsync(userId, Enum.ThumbnailType.AvatarBust, Enum.ThumbnailSize.Size420x420)
    end)
    
    if thumbnail then
        charThumbnail.Image = thumbnail
    else
        charThumbnail.Image = "rbxasset://textures/ui/GuiImagePlaceholder.png"
    end
    
    charNameLabel.Text = username
    
    -- Vérifier si le joueur est dans le serveur
    local targetPlayer = Players:FindFirstChild(username)
    if targetPlayer then
        local ping = math.floor((targetPlayer:GetNetworkPing() or 0) * 1000)
        local hp, maxHp = 100, 100
        local speed = 16
        if targetPlayer.Character then
            local hum = targetPlayer.Character:FindFirstChildOfClass("Humanoid")
            if hum then
                hp = math.floor(hum.Health)
                maxHp = math.floor(hum.MaxHealth)
                speed = math.floor(hum.WalkSpeed)
            end
        end
        charInfoLabel.Text = "In-Server | " .. ping .. "ms | " .. hp .. "/" .. maxHp .. " | Speed:" .. speed
    else
        charInfoLabel.Text = "Not in this server | ID: " .. userId .. " | Click 'Apply Skin' to copy"
    end
end

local dragging = false
local dragStart, startPos

titleBar.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 then
        dragging = true
        dragStart = input.Position
        startPos = mainFrame.Position
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if dragging and input.UserInputType == Enum.UserInputType.MouseMovement then
        local delta = input.Position - dragStart
        mainFrame.Position = UDim2.new(
            startPos.X.Scale, startPos.X.Offset + delta.X,
            startPos.Y.Scale, startPos.Y.Offset + delta.Y
        )
    end
end)

UserInputService.InputEnded:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 then
        dragging = false
    end
end)

UserInputService.InputBegan:Connect(function(input, gp)
    if gp then return end
    
    if guiToggleKey then
        if (input.UserInputType == Enum.UserInputType.Keyboard and input.KeyCode == guiToggleKey) or
           (input.UserInputType == guiToggleKey) then
            toggleGui()
        end
    end
    
    if input.UserInputType == Enum.UserInputType.Keyboard then
        if speedKey and input.KeyCode == speedKey then speedTog.SetState(not speedTog.GetState()) end
        if jumpKey and input.KeyCode == jumpKey then jumpTog.SetState(not jumpTog.GetState()) end
        if flyKey and input.KeyCode == flyKey then flyTog.SetState(not flyTog.GetState()) end
        if invisKey and input.KeyCode == invisKey then invisTog.SetState(not invisTog.GetState()) end
    elseif input.UserInputType == Enum.UserInputType.MouseButton1 or
           input.UserInputType == Enum.UserInputType.MouseButton2 or
           input.UserInputType == Enum.UserInputType.MouseButton3 then
        if speedKey and input.UserInputType == speedKey then speedTog.SetState(not speedTog.GetState()) end
        if jumpKey and input.UserInputType == jumpKey then jumpTog.SetState(not jumpTog.GetState()) end
        if flyKey and input.UserInputType == flyKey then flyTog.SetState(not flyTog.GetState()) end
        if invisKey and input.UserInputType == invisKey then invisTog.SetState(not invisTog.GetState()) end
    end
end)
