-- ============================================================
local function ApplyAnimations(Animations, Character)
    local Animate = Character:FindFirstChild('Animate') or Character:WaitForChild('Animate', 5)
    if not Animate then return end
    
    for Animation, Ids in next, Animations do
        local AnimObject = Animate:FindFirstChild(Animation)
        if not AnimObject then continue end
        
        for i, id in next, Ids do
            local AnimationObject
            if Animation == 'idle' then
                AnimationObject = AnimObject:FindFirstChild('Animation'..tostring(i))
            elseif Animation == 'swim' and i == 2 then 
                local swimidle = Animate:FindFirstChild('swimidle')
                AnimationObject = swimidle and swimidle:FindFirstChild('SwimIdle')
            else
                local firststchar = string.upper(string.sub(Animation,1,1))
                local restchar = string.sub(Animation,2,#Animation)
                local fullstr = firststchar..restchar
                AnimationObject = AnimObject:FindFirstChild(fullstr..'Anim') or AnimObject:FindFirstChild(fullstr)
            end
            
            if AnimationObject and id then
                AnimationObject.AnimationId = "rbxassetid://"..tostring(id)
            end
        end
    end
end

-- ROBUST FLING SYSTEM (KILASIK's Method)
-- ============================================================
local function SkidFling(TargetPlayer)
    local Character = player.Character
    local Humanoid = Character and Character:FindFirstChildOfClass("Humanoid")
    local RootPart = Humanoid and Humanoid.RootPart
    local TCharacter = TargetPlayer.Character
    if not TCharacter or not Character or not RootPart then return end
    
    local THumanoid = TCharacter:FindFirstChildOfClass("Humanoid")
    local TRootPart = THumanoid and THumanoid.RootPart
    if not THumanoid or not TRootPart then return end
    if THumanoid.Sit then return end
    
    local OldPos = RootPart.CFrame
    local FPDH = Workspace.FallenPartsDestroyHeight
    Workspace.FallenPartsDestroyHeight = 0/0 -- Disable part destruction
    
    local BV = Instance.new("BodyVelocity")
    BV.Parent = RootPart
    BV.Velocity = Vector3.new(0, 0, 0)
    BV.MaxForce = Vector3.new(9e9, 9e9, 9e9)
    
    Humanoid:SetStateEnabled(Enum.HumanoidStateType.Seated, false)
    Character.PrimaryPart = RootPart
    
    local flingActive = true
    task.delay(2, function() flingActive = false end)
    
    local Angle = 0
    while flingActive and TCharacter and TCharacter.Parent and TRootPart do
        Angle = Angle + 100
        
        -- Chaotic movement in all directions
        local pos1 = CFrame.new(TRootPart.Position) * CFrame.new(0, 1.5, 0) * CFrame.Angles(math.rad(Angle), 0, 0)
        RootPart.CFrame = pos1
        Character:SetPrimaryPartCFrame(pos1)
        RootPart.Velocity = Vector3.new(9e7, 9e8, 9e7)
        RootPart.RotVelocity = Vector3.new(9e8, 9e8, 9e8)
        task.wait()
        
        local pos2 = CFrame.new(TRootPart.Position) * CFrame.new(0, -1.5, 0) * CFrame.Angles(math.rad(Angle), 0, 0)
        RootPart.CFrame = pos2
        Character:SetPrimaryPartCFrame(pos2)
        RootPart.Velocity = Vector3.new(9e7, 9e8, 9e7)
        RootPart.RotVelocity = Vector3.new(9e8, 9e8, 9e8)
        task.wait()
    end
    
    BV:Destroy()
    Humanoid:SetStateEnabled(Enum.HumanoidStateType.Seated, true)
    Workspace.FallenPartsDestroyHeight = FPDH
    
    -- Reset player position and velocity
    RootPart.CFrame = OldPos * CFrame.new(0, 0.5, 0)
    Character:SetPrimaryPartCFrame(OldPos * CFrame.new(0, 0.5, 0))
    Humanoid:ChangeState(Enum.HumanoidStateType.GettingUp)
    for _, part in pairs(Character:GetChildren()) do
        if part:IsA("BasePart") then
            part.Velocity, part.RotVelocity = Vector3.new(), Vector3.new()
        end
    end
end

            checkWalls = checkWalls, checkTeam = checkTeam, espEnabled = espEnabled,
            showNames = showNames, showHealth = showHealth, speedEnabled = speedEnabled,

local function isAimKeyHeld()
    if typeof(aimKey) == "EnumItem" then
        if aimKey.EnumType == Enum.KeyCode then return UserInputService:IsKeyDown(aimKey)
        elseif aimKey.EnumType == Enum.UserInputType then return UserInputService:IsMouseButtonPressed(aimKey) end
    end
    return false
end

RunService.RenderStepped:Connect(function()
    fovCircle.Position = UserInputService:GetMouseLocation()
    
    if not aimEnabled or not isAimKeyHeld() then 
        lockedTarget = nil
        return 
    end

    local camera = Workspace.CurrentCamera
    if not camera then return end

    -- Sticky aimlock: if we have a locked target, check if it's still valid
    if lockedTarget and lockedTarget.Character then
        local hum = lockedTarget.Character:FindFirstChildOfClass("Humanoid")
        if hum and hum.Health > 0 then
            local targetPart = lockedTarget.Character:FindFirstChild("Head") or lockedTarget.Character:FindFirstChild("HumanoidRootPart")
            if targetPart then
                local valid = true
                if checkTeam and player.Team and lockedTarget.Team and player.Team == lockedTarget.Team then valid = false end
                
                if checkWalls and valid then
                    local origin = camera.CFrame.Position
                    local direction = (targetPart.Position - origin).Unit
                    local distance = (targetPart.Position - origin).Magnitude
                    local params = RaycastParams.new()
                    params.FilterDescendantsInstances = {player.Character, lockedTarget.Character}
                    params.FilterType = Enum.RaycastFilterType.Exclude
                    local rayResult = Workspace:Raycast(origin, direction * distance, params)
                    if rayResult and not rayResult.Instance:IsDescendantOf(lockedTarget.Character) then 
                        valid = false 
                    end
                end
                
                if valid then
                    local targetPos = targetPart.Position
                    if smoothnessVal <= 1 then
                        camera.CFrame = CFrame.new(camera.CFrame.Position, targetPos)
                    else
                        local factor = (21 - smoothnessVal) / 100
                        camera.CFrame = camera.CFrame:Lerp(CFrame.new(camera.CFrame.Position, targetPos), factor)
                    end
                    return -- Stick to this target!
                end
            end
        end
    end
    
    -- If no valid locked target, search for a new one
    lockedTarget = nil
    local mousePos = UserInputService:GetMouseLocation()
    local closestTarget = nil
    local shortestDistance = fovRadius

    for _, p in ipairs(Players:GetPlayers()) do
        if p == player then continue end
        
        if #targetedPlayers > 0 then
            local isTargeted = false
            for _, tName in ipairs(targetedPlayers) do
                if tName == p.Name then isTargeted = true break end
            end
            if not isTargeted then continue end
        end
        
        if checkTeam and p.Team and player.Team and p.Team == player.Team then continue end
        
        local char = p.Character
        if not char then continue end
        
        local hum = char:FindFirstChildOfClass("Humanoid")
        if not hum or hum.Health <= 0 then continue end

        local targetPart = char:FindFirstChild("Head") or char:FindFirstChild("HumanoidRootPart")
        if not targetPart then continue end

        if checkWalls then
            local origin = camera.CFrame.Position
            local direction = (targetPart.Position - origin).Unit
            local distance = (targetPart.Position - origin).Magnitude
            local params = RaycastParams.new()
            params.FilterDescendantsInstances = {player.Character, char}
            params.FilterType = Enum.RaycastFilterType.Exclude
            local rayResult = Workspace:Raycast(origin, direction * distance, params)
            if rayResult and not rayResult.Instance:IsDescendantOf(char) then continue end
        end

        local screenPos, onScreen = camera:WorldToViewportPoint(targetPart.Position)
        if onScreen then
            local dist = (Vector2.new(screenPos.X, screenPos.Y) - mousePos).Magnitude
            if dist < shortestDistance then
                shortestDistance = dist
                closestTarget = p
            end
        end
    end

    if closestTarget then
        lockedTarget = closestTarget
        local tp = lockedTarget.Character:FindFirstChild("Head") or lockedTarget.Character:FindFirstChild("HumanoidRootPart")
        if tp then
            local targetPos = tp.Position
            if smoothnessVal <= 1 then
                camera.CFrame = CFrame.new(camera.CFrame.Position, targetPos)
            else
                local factor = (21 - smoothnessVal) / 100
                camera.CFrame = camera.CFrame:Lerp(CFrame.new(camera.CFrame.Position, targetPos), factor)
            end
        end
    end
end)

local function cleanupESP(p)
    local data = activeESPs[p]
    if data then
        pcall(function()
            if data.Highlight then data.Highlight:Destroy() end
            if data.Billboard then data.Billboard:Destroy() end
        end)
        activeESPs[p] = nil
    end
end

local function ensureESP(p)
    if p == player or not espEnabled then return end
    local char = p.Character
    if not char then return end
    
    local needsRecreation = false
    if activeESPs[p] then
        local data = activeESPs[p]
        if not data.Highlight or not data.Highlight.Parent or data.Highlight.Adornee ~= char then needsRecreation = true end
        if not data.Billboard or not data.Billboard.Parent or data.Billboard.Adornee ~= char then needsRecreation = true end
        
        if needsRecreation then
            cleanupESP(p)
        else
            data.Highlight.FillColor = espColor
            data.Highlight.OutlineColor = espColor
            data.Highlight.Enabled = true
            local nl = data.Billboard:FindFirstChild("NL")
            local hl = data.Billboard:FindFirstChild("HL")
            if nl then nl.Visible = showNames; nl.TextColor3 = espColor end
            if hl then hl.Visible = showHealth; hl.TextColor3 = espColor end
            return
        end
    end

    local root = char:FindFirstChild("HumanoidRootPart")
    local head = char:FindFirstChild("Head")
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not root then return end

    local data = {}
    local hl = Instance.new("Highlight")
    hl.Name = "Fucker_Highlight_" .. p.Name
    hl.Adornee = char
    hl.FillColor = espColor
    hl.FillTransparency = 0.75
    hl.OutlineColor = espColor
    hl.OutlineTransparency = 0
    hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
    hl.Parent = char
    hl.Enabled = true
    data.Highlight = hl

    local bb = Instance.new("BillboardGui")
    bb.Name = "Fucker_Billboard_" .. p.Name
    bb.Adornee = head or root
    bb.Size = UDim2.new(0, 200, 0, 60)
    bb.StudsOffset = Vector3.new(0, 3.5, 0)
    bb.AlwaysOnTop = true
    bb.Enabled = true
    bb.Parent = char

    local nl = Instance.new("TextLabel")
    nl.Name = "NL"
    nl.Size = UDim2.new(1, 0, 0.5, 0)
    nl.BackgroundTransparency = 1
    nl.Text = p.Name
    nl.TextColor3 = espColor
    nl.TextSize = 16
    nl.Font = Enum.Font.GothamBold
    nl.TextStrokeTransparency = 0.6
    nl.Visible = showNames
    nl.Parent = bb

    local hlbl = Instance.new("TextLabel")
    hlbl.Name = "HL"
    hlbl.Size = UDim2.new(1, 0, 0.5, 0)
    hlbl.Position = UDim2.new(0, 0, 0.5, 0)
    hlbl.BackgroundTransparency = 1
    hlbl.Text = hum and math.floor(hum.Health).."/"..math.floor(hum.MaxHealth) or "100/100"
    hlbl.TextColor3 = espColor
    hlbl.TextSize = 14
    hlbl.Font = Enum.Font.Gotham
    hlbl.TextStrokeTransparency = 0.6
    hlbl.Visible = showHealth
    hlbl.Parent = bb

    data.Billboard = bb
    activeESPs[p] = data
end

local function applyMovement()
    local char = player.Character
    if not char then return end
    local hum = char:FindFirstChildOfClass("Humanoid")
    if hum and not flyEnabled then 
        hum.WalkSpeed = speedEnabled and walkSpeedVal or 16 
    end
end

function updateFly()
    local char = player.Character
    if not char then return end
    local root = char:FindFirstChild("HumanoidRootPart")
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not root or not hum then return end

    if flyEnabled then
        hum.PlatformStand = true
        if not flyConn then
            flyConn = RunService.RenderStepped:Connect(function()
                if not flyEnabled or not player.Character then updateFly(); return end
                local currentRoot = player.Character:FindFirstChild("HumanoidRootPart")
                if not currentRoot then return end
                
                local cam = Workspace.CurrentCamera
                local moveDir = Vector3.new(0, 0, 0)
                if UserInputService:IsKeyDown(Enum.KeyCode.W) then moveDir = moveDir + cam.CFrame.LookVector end
                if UserInputService:IsKeyDown(Enum.KeyCode.S) then moveDir = moveDir - cam.CFrame.LookVector end
                if UserInputService:IsKeyDown(Enum.KeyCode.A) then moveDir = moveDir - cam.CFrame.RightVector end
                if UserInputService:IsKeyDown(Enum.KeyCode.D) then moveDir = moveDir + cam.CFrame.RightVector end
                if UserInputService:IsKeyDown(Enum.KeyCode.Space) then moveDir = moveDir + Vector3.new(0, 1, 0) end
                if UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) or UserInputService:IsKeyDown(Enum.KeyCode.LeftShift) then 
                    moveDir = moveDir - Vector3.new(0, 1, 0) 
                end
                if moveDir.Magnitude > 0 then moveDir = moveDir.Unit end
                currentRoot.AssemblyLinearVelocity = moveDir * flySpeedVal
                currentRoot.RotVelocity = Vector3.new(0, 0, 0)
            end)
        end
    else
        if flyConn then flyConn:Disconnect(); flyConn = nil end
        hum.PlatformStand = false
        applyMovement()
    end
end

    if antiFlingEnabled and player.Character and currentTime - antiFlingLastCheck > 0.1 then
        antiFlingLastCheck = currentTime
        
        local root = player.Character:FindFirstChild("HumanoidRootPart")

                root.AssemblyLinearVelocity = Vector3.new(0, 0, 0)

                    local pRoot = p.Character:FindFirstChild("HumanoidRootPart")
                    if pRoot then pRoot.CanCollide = false end
                end

        end
    end
    
    if noclipEnabled and player.Character then
        for _, part in ipairs(player.Character:GetDescendants()) do
            if part:IsA("BasePart") then part.CanCollide = false end
        end
    end
end)

UserInputService.InputBegan:Connect(function(input, gameProcessed)
    if gameProcessed then return end
    
    local function isKeyMatch(keybind, input)
        if not keybind then return false end
        if typeof(keybind) == "EnumItem" then
            if keybind.EnumType == Enum.KeyCode and input.KeyCode == keybind then return true end
            if keybind.EnumType == Enum.UserInputType and input.UserInputType == keybind then return true end
        end
        return false
    end
    
    if isKeyMatch(speedKeybind, input) then
        speedEnabled = not speedEnabled
        speedToggle.setState(speedEnabled)
        applyMovement()
    end
    
    if isKeyMatch(flyKeybind, input) then
        flyEnabled = not flyEnabled
        flyToggle.setState(flyEnabled)
        updateFly()
    end
    
    if isKeyMatch(jumpKeybind, input) then
        infJumpEnabled = not infJumpEnabled
        infJumpToggle.setState(infJumpEnabled)
    end
    
    if isKeyMatch(noclipKeybind, input) then
        noclipEnabled = not noclipEnabled
        noclipToggle.setState(noclipEnabled)
    end
end)
